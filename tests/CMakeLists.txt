cmake_minimum_required(VERSION 3.15)
project(green_stripe_jsfx_tests LANGUAGES C CXX ASM)
if(NOT YSFX_SOURCE_DIR)
    message(FATAL_ERROR "Set -DYSFX_SOURCE_DIR=/absolute/path/to/pinned/ysfx")
endif()
set(YSFX_PLUGIN OFF CACHE BOOL "" FORCE)
set(YSFX_GFX OFF CACHE BOOL "" FORCE)
set(YSFX_TESTS OFF CACHE BOOL "" FORCE)
set(YSFX_TOOLS OFF CACHE BOOL "" FORCE)
# The pinned ysfx checksum target uses CMAKE_SOURCE_DIR and fails when embedded.
# Verify that same upstream assembly checksum here before disabling only its target.
file(READ "${YSFX_SOURCE_DIR}/thirdparty/WDL/source/WDL/eel2/asm-nseel-x64-sse.asm" YSFX_ASM)
string(REPLACE "\r\n" "\n" YSFX_ASM "${YSFX_ASM}")
string(SHA512 YSFX_ASM_SHA "${YSFX_ASM}")
file(READ "${YSFX_SOURCE_DIR}/sources/eel2-gas/sources/ref-hash-sha512.txt" YSFX_EXPECTED_SHA)
string(STRIP "${YSFX_EXPECTED_SHA}" YSFX_EXPECTED_SHA)
if(NOT YSFX_ASM_SHA STREQUAL YSFX_EXPECTED_SHA)
    message(FATAL_ERROR "ysfx upstream assembly checksum mismatch")
endif()
set(YSFX_SKIP_CHECKSUM ON CACHE BOOL "Already verified by parent project" FORCE)
add_subdirectory("${YSFX_SOURCE_DIR}" ysfx)
add_executable(jsfx_parity jsfx_parity.cpp)
add_executable(benchmark_jsfx benchmark_jsfx.cpp)
target_include_directories(jsfx_parity PRIVATE ../src "${YSFX_SOURCE_DIR}/include")
target_link_libraries(jsfx_parity PRIVATE ysfx)
target_include_directories(benchmark_jsfx PRIVATE "${YSFX_SOURCE_DIR}/include")
target_link_libraries(benchmark_jsfx PRIVATE ysfx)
set_target_properties(benchmark_jsfx PROPERTIES CXX_STANDARD 17)
set_target_properties(jsfx_parity PROPERTIES CXX_STANDARD 17)
if(NOT MSVC)
    target_compile_options(jsfx_parity PRIVATE -fno-fast-math -ffp-contract=off)
endif()
