cmake_minimum_required(VERSION 3.15)
project(GreenStripe76 VERSION 0.2.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GS76_BUILD_TESTS "Build signal acceptance tests" ON)
add_library(green-stripe-76 MODULE src/lv2_plugin.cpp)
target_include_directories(green-stripe-76 PRIVATE src)
set_target_properties(green-stripe-76 PROPERTIES PREFIX "" CXX_VISIBILITY_PRESET hidden)
if(NOT MSVC)
    target_compile_options(green-stripe-76 PRIVATE -fno-fast-math -ffp-contract=off -fno-exceptions -fno-rtti -Wall -Wextra)
    if(UNIX AND NOT APPLE)
        target_link_options(green-stripe-76 PRIVATE -Wl,--no-undefined)
    endif()
endif()
add_custom_command(TARGET green-stripe-76 POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_SOURCE_DIR}/lv2/green-stripe-76.lv2" "${CMAKE_BINARY_DIR}/green-stripe-76.lv2"
    COMMAND ${CMAKE_COMMAND} -E copy "$<TARGET_FILE:green-stripe-76>" "${CMAKE_BINARY_DIR}/green-stripe-76.lv2/green-stripe-76${CMAKE_SHARED_MODULE_SUFFIX}")
file(GLOB_RECURSE GS76_BUNDLE_ASSETS CONFIGURE_DEPENDS "${CMAKE_SOURCE_DIR}/lv2/green-stripe-76.lv2/*")
set_property(TARGET green-stripe-76 APPEND PROPERTY LINK_DEPENDS ${GS76_BUNDLE_ASSETS})
if(GS76_BUILD_TESTS)
    enable_testing()
    add_executable(dsp_tests tests/dsp_tests.cpp)
    add_executable(transitions tests/transitions.cpp)
    target_include_directories(dsp_tests PRIVATE src)
    target_include_directories(transitions PRIVATE src)
    if(NOT MSVC)
        target_compile_options(dsp_tests PRIVATE -fno-fast-math -ffp-contract=off)
        target_compile_options(transitions PRIVATE -fno-fast-math -ffp-contract=off)
    endif()
    add_test(NAME signal_tests COMMAND dsp_tests)
    add_test(NAME transition_tests COMMAND transitions)
endif()
if(UNIX AND NOT APPLE)
    install(DIRECTORY "${CMAKE_BINARY_DIR}/green-stripe-76.lv2" DESTINATION lib/lv2)
endif()
